rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function userProfile() {
      return get(
        /databases/$(database)/documents/users/$(request.auth.uid)
      ).data;
    }

    function hasProfile() {
      return signedIn() &&
        exists(
          /databases/$(database)/documents/users/$(request.auth.uid)
        );
    }

    function isManager() {
      return hasProfile() && userProfile().role == "gestor";
    }

    function isTeacher() {
      return hasProfile() && userProfile().role == "professor";
    }

    function isStudent() {
      return hasProfile() && userProfile().role == "aluno";
    }

    function isBootstrapManager(uid) {
      return signedIn()
        && request.auth.uid == uid
        && request.auth.token.email == "admin@fitcontrol.com";
    }

    match /users/{uid} {
      allow read: if signedIn() &&
        (request.auth.uid == uid || isManager());

      allow create: if
        (
          isBootstrapManager(uid)
          && request.resource.data.role == "gestor"
        )
        || isManager();

      allow update, delete: if isManager();
    }

    match /academyData/{key} {
      allow read: if isManager()
        || (
          isTeacher()
          && key in [
            "fitcontrol_alunos",
            "fitcontrol_professores",
            "fitcontrol_fichas_treino",
            "champion_team_graduacoes",
            "champion_team_regras_graduacao",
            "champion_team_indicacoes_faixa"
          ]
        );

      allow create, update: if isManager()
        || (
          isTeacher()
          && key in [
            "fitcontrol_fichas_treino",
            "champion_team_indicacoes_faixa"
          ]
        );

      allow delete: if isManager();
    }

    match /studentViews/{uid} {
      allow read: if signedIn()
        && request.auth.uid == uid
        && isStudent();

      allow create, update, delete: if isManager();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
